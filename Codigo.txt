const SPREADSHEET_ID = "1e6sCQi0g3jlhIkU1OVCjqoQ7CorT854X48t_nst_U84";
const TEMPLATE_DOC_ID = "1Knnv5Uas66UC5Wsx4ir2JAFjbhROVZoKH9mw0N10RyM";

function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('Sistema Maestro-Detalle de Devoluciones')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function getInitialData() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);

  const sheetFunc = ss.getSheetByName("FUNCIONARIOS");
  const lastRowFunc = sheetFunc.getLastRow();
  const funcionarios = lastRowFunc > 1
    ? sheetFunc.getRange(2, 1, lastRowFunc - 1, 1).getValues().flat().filter(Boolean)
    : [];

  const sheetMotivos = ss.getSheetByName("MOTIVOS");
  const lastRowMotivos = sheetMotivos.getLastRow();
  const motivos = lastRowMotivos > 1
    ? sheetMotivos.getRange(2, 1, lastRowMotivos - 1, 1).getValues().flat().filter(Boolean)
    : [];

  const sheetDev = ss.getSheetByName("DEVOLUCIONES");
  const devData = sheetDev.getDataRange().getValues();
  devData.shift();
  const devoluciones = devData.map(row => ({
    consecutivo: row[0],
    fecha: row[1] instanceof Date
      ? Utilities.formatDate(row[1], Session.getScriptTimeZone(), "yyyy-MM-dd")
      : row[1],
    funcionario: row[2],
    placa: row[3],
    documento: row[4],
    ciudadano: row[5]
  })).reverse();

  return { funcionarios, motivos, devoluciones };
}

function getDetalleDevolucion(id) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheetDet = ss.getSheetByName("DEVOLUCIONES_DETALLES");
  const data = sheetDet.getDataRange().getValues();
  data.shift();
  return data
    .filter(row => String(row[1]) === String(id))
    .map(row => ({
      nro: row[0],
      id: row[1],
      motivo: row[2],
      observacion: row[3] || ""
    }));
}

function guardarDevolucion(master, detalles) {
  const lock = LockService.getScriptLock();
  lock.waitLock(10000);

  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheetDev = ss.getSheetByName("DEVOLUCIONES");
    const sheetDet = ss.getSheetByName("DEVOLUCIONES_DETALLES");

    const devData = sheetDev.getDataRange().getValues();
    let maxId = 0;
    for (let i = 1; i < devData.length; i++) {
      let val = parseInt(devData[i][0]);
      if (!isNaN(val) && val > maxId) maxId = val;
    }
    const newId = maxId + 1;
    const fechaHoy = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");

    sheetDev.appendRow([
      newId, fechaHoy, master.funcionario,
      master.placa ? master.placa.toUpperCase() : "",
      master.documento,
      master.ciudadano ? master.ciudadano.toUpperCase() : ""
    ]);

    if (detalles && detalles.length > 0) {
      const lastRowDet = sheetDet.getLastRow();
      let currentNro = lastRowDet > 1 ? lastRowDet : 1;
      const rowsToAppend = detalles.map(item => [
        currentNro++, newId, item.motivo, item.observacion || ""
      ]);
      sheetDet.getRange(sheetDet.getLastRow() + 1, 1, rowsToAppend.length, 4)
        .setValues(rowsToAppend);
    }

    return newId;
  } finally {
    lock.releaseLock();
  }
}

/*************************************************
 * GENERAR PDF DE DEVOLUCIÓN
 *************************************************/
function generarPDF(consecutivoId) {

  let copia = null;
  let doc = null;

  try {

    //==========================================
    // OBTENER DATOS DEL MAESTRO
    //==========================================

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheetDev = ss.getSheetByName("DEVOLUCIONES");

    if (!sheetDev) {
      throw new Error("No existe la hoja DEVOLUCIONES.");
    }

    const datos = sheetDev.getDataRange().getValues();

    let master = null;

    for (let i = 1; i < datos.length; i++) {

      if (String(datos[i][0]) === String(consecutivoId)) {

        master = {
          id: datos[i][0],
          fecha: datos[i][1] instanceof Date
            ? Utilities.formatDate(
                datos[i][1],
                Session.getScriptTimeZone(),
                "yyyy-MM-dd"
              )
            : datos[i][1],
          funcionario: datos[i][2],
          placa: datos[i][3],
          documento: datos[i][4],
          ciudadano: datos[i][5]
        };

        break;
      }

    }

    if (!master) {
      throw new Error("No existe la devolución " + consecutivoId);
    }

    //==========================================
    // DETALLE
    //==========================================

    const detalles = getDetalleDevolucion(consecutivoId);

    //==========================================
    // CREAR COPIA DE LA PLANTILLA
    //==========================================

    const plantilla = DriveApp.getFileById(TEMPLATE_DOC_ID);

    copia = plantilla.makeCopy("DEVOLUCION_" + master.id);

    doc = DocumentApp.openById(copia.getId());

    const body = doc.getBody();

    //==========================================
    // REEMPLAZAR MARCADORES
    //==========================================

    body.replaceText("\\{\\{ID\\}\\}", String(master.id));
    body.replaceText("\\{\\{FECHA\\}\\}", String(master.fecha));
    body.replaceText("\\{\\{FUNCIONARIO\\}\\}", String(master.funcionario));
    body.replaceText("\\{\\{PLACA\\}\\}", String(master.placa || ""));
    body.replaceText("\\{\\{CEDULA\\}\\}", String(master.documento || ""));
    body.replaceText("\\{\\{CIUDADANO\\}\\}", String(master.ciudadano || ""));

    //==========================================
    // INSERTAR TABLA
    //==========================================

    const encontrado = body.findText("\\{\\{TABLA\\}\\}");

    if (encontrado) {

      const elemento = encontrado.getElement();
      const parrafo = elemento.getParent().asParagraph();
      const indice = body.getChildIndex(parrafo);

      const datosTabla = [];

      datosTabla.push([
        "MOTIVO",
        "OBSERVACIÓN"
      ]);

      detalles.forEach(function(det) {

        datosTabla.push([
          det.motivo || "",
          det.observacion || ""
        ]);

      });

      // Insertar la tabla
      body.insertTable(indice, datosTabla);

      // Limpiar marcador SIN eliminar el párrafo
      parrafo.clear();

    }

    //==========================================
    // GUARDAR
    //==========================================

    doc.saveAndClose();

    Utilities.sleep(1000);

    //==========================================
    // PDF
    //==========================================

    const pdf = copia.getBlob().getAs(MimeType.PDF);

    pdf.setName(
      "Devolucion_" +
      (master.placa || "SIN_PLACA") +
      "_" +
      master.id +
      ".pdf"
    );

    return {
      filename: pdf.getName(),
      base64: Utilities.base64Encode(pdf.getBytes())
    };

  } catch (err) {

    throw new Error("Error al generar el PDF: " + err.message);

  } finally {

    try {

      if (doc) {
        doc.saveAndClose();
      }

    } catch (e) {}

    try {

      if (copia) {
        copia.setTrashed(true);
      }

    } catch (e) {}

  }

}
